Detailed Backend Flow (Java)</h3><h4>5.1 New Endpoint – Process Widget File</h4><p>New endpoint in <code>FileMapperController</code>:</p><p><br /></p><ac:structured-macro ac:name="code" ac:schema-version="1" ac:macro-id="9dedd0a9-cdda-457c-aa2c-8f2bb26d61fe"><ac:plain-text-body><![CDATA[@PostMapping(

@CventAuthorization(scopes = {"PASSKEY_USER"})
public ResponseEntity uploadWidgetFile(
    @Valid @PathVariable("participantId") Long participantId,
    @Valid @PathVariable("widgetFileId") String widgetFileId) {
    LOG.info("Processing widget file {} for participant {}", widgetFileId, participantId);

    try {
        // 1. Download sanitized file and metadata from passkey-file-uploader
        SanitizedFile sanitized = fileUploaderClient.downloadSanitizedFile(widgetFileId);

        String fileNameSanitized =
            ParsingUtil.sanitizeFilenameString(sanitized.getOriginalFileName());

        // 2. Upload to our S3 via existing FileMapperService
        Object response = fileMapperService.uploadFile(
            fileNameSanitized,
            sanitized.getInputStream(),
            participantId
        );

        return ResponseEntity.ok(response);
    } catch (InvalidFileTypeException e) {
        return ResponseEntity.status(HttpStatus.BAD_REQUEST)
            .body("Unsupported file type.");
    } catch (FileNotFoundInUploaderException e) {
        return ResponseEntity.status(HttpStatus.NOT_FOUND)
            .body("Widget file not found.");
    } catch (Exception e) {
        LOG.error("Failed to process widget file {} for participant {}: {}",
            widgetFileId, participantId, e.getMessage());
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
            .body("Failed to process widget file.");
    }
}
