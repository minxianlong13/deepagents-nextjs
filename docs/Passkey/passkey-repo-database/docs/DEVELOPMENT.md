```sql
-- Use uppercase for SQL keywords

-- Use meaningful aliases
SELECT h.hotel_id, h.hotel_name, c.country_name

-- Format complex queries for readability
SELECT 
    h.hotel_id,
    h.hotel_name,
    COUNT(r.reservation_id) AS total_reservations,
    SUM(r.total_amount) AS total_revenue
FROM COMMON.HOTELS h
LEFT JOIN BRIDGE.RESERVATIONS r ON h.hotel_id = r.hotel_id
WHERE h.status = 'ACTIVE'
AND r.created_date >= TRUNC(SYSDATE) - 30
GROUP BY h.hotel_id, h.hotel_name
HAVING COUNT(r.reservation_id) > 0
ORDER BY total_revenue DESC;
```

##### Develop Database Changes
```sql
-- Example: TXN/FEATURES/GROUP-XXXX/001_create_hotel_amenities.sql
CREATE TABLE COMMON.HOTEL_AMENITIES (
    amenity_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    hotel_id NUMBER NOT NULL,
    amenity_type VARCHAR2(50) NOT NULL,
    amenity_name VARCHAR2(255) NOT NULL,
    description CLOB,
    is_available CHAR(1) DEFAULT 'Y' CHECK (is_available IN ('Y', 'N')),
    created_date DATE DEFAULT SYSDATE NOT NULL,
    created_by NUMBER NOT NULL,
    modified_date DATE,
    modified_by NUMBER,
    CONSTRAINT fk_hotel_amenities_hotel 
        FOREIGN KEY (hotel_id) REFERENCES COMMON.HOTELS(hotel_id)
)
/

-- Create index for performance
CREATE INDEX idx_hotel_amenities_hotel_id 
ON COMMON.HOTEL_AMENITIES(hotel_id, is_available)
/

-- Add comments for documentation
COMMENT ON TABLE COMMON.HOTEL_AMENITIES IS 'Hotel amenities and facilities';
COMMENT ON COLUMN COMMON.HOTEL_AMENITIES.amenity_type IS 'Type of amenity (POOL, GYM, WIFI, etc.)';
```

## Coding Standards

### SQL Coding Standards

#### Naming Conventions
```sql
-- Tables: Singular nouns, uppercase
HOTELS, EVENTS, RESERVATIONS

-- Columns: Descriptive names, lowercase with underscores
hotel_id, hotel_name, created_date

-- Indexes: Descriptive with prefix
idx_hotels_city_status
pk_hotels
fk_events_hotel_id

-- Constraints: Descriptive with prefix
chk_hotels_status
fk_reservations_hotel_id
uk_hotels_code

-- Sequences: Table name + _seq
hotel_seq, event_seq

-- Procedures/Functions: Descriptive action names
create_hotel, get_hotel_details, update_reservation_status
```

#### Code Structure
```sql
-- File header template
/*
 * File: 001_create_hotel_amenities.sql
 * Feature: GROUP-XXXX
 * Author: Developer Name
 * Date: YYYY-MM-DD
 * Description: Create hotel amenities table and related objects
 * 
 * Dependencies:
 * - COMMON.HOTELS table must exist
 * 
 * Rollback: 001_drop_hotel_amenities.sql
 */

-- Main SQL content with proper formatting
CREATE TABLE COMMON.HOTEL_AMENITIES (
    amenity_id NUMBER GENERATED BY DEFAULT AS IDENTITY,
    hotel_id NUMBER NOT NULL,
    amenity_type VARCHAR2(50) NOT NULL,
    -- Additional columns...
    
    -- Constraints
    CONSTRAINT pk_hotel_amenities PRIMARY KEY (amenity_id),
    CONSTRAINT fk_hotel_amenities_hotel 
        FOREIGN KEY (hotel_id) REFERENCES COMMON.HOTELS(hotel_id),
    CONSTRAINT chk_hotel_amenities_available 
        CHECK (is_available IN ('Y', 'N'))
)
/

-- Always end with forward slash for PL/SQL blocks
```

#### Error Handling
```sql
-- Stored procedure error handling template
CREATE OR REPLACE PROCEDURE create_hotel_amenity(
    p_hotel_id IN NUMBER,
    p_amenity_type IN VARCHAR2,
    p_amenity_name IN VARCHAR2,
    p_amenity_id OUT NUMBER
) IS
    l_count NUMBER;
BEGIN
    -- Input validation
    IF p_hotel_id IS NULL THEN
        RAISE_APPLICATION_ERROR(-20001, 'Hotel ID cannot be null');
    END IF;
    
    IF p_amenity_type IS NULL OR p_amenity_name IS NULL THEN
        RAISE_APPLICATION_ERROR(-20002, 'Amenity type and name are required');
    END IF;
    
    -- Business logic validation
    SELECT COUNT(*)
    INTO l_count
    FROM COMMON.HOTELS
    WHERE hotel_id = p_hotel_id
    AND status = 'ACTIVE';
    
    IF l_count = 0 THEN
        RAISE_APPLICATION_ERROR(-20003, 'Hotel not found or inactive');
    END IF;
    
    -- Main logic
    INSERT INTO COMMON.HOTEL_AMENITIES (
        hotel_id, amenity_type, amenity_name, created_by
    ) VALUES (
        p_hotel_id, p_amenity_type, p_amenity_name, 
        NVL(SYS_CONTEXT('USERENV', 'CLIENT_IDENTIFIER'), USER)
    ) RETURNING amenity_id INTO p_amenity_id;
    
    COMMIT;
    
EXCEPTION
    WHEN DUP_VAL_ON_INDEX THEN
        RAISE_APPLICATION_ERROR(-20004, 'Duplicate amenity for this hotel');
    WHEN OTHERS THEN
        ROLLBACK;
        -- Log error details
        INSERT INTO COMMON.ERROR_LOG (
            error_code, error_message, procedure_name, 
            error_date, user_name
        ) VALUES (
            SQLCODE, SQLERRM, 'create_hotel_amenity',
            SYSDATE, USER
        );
        COMMIT;
        RAISE;
END create_hotel_amenity;
/
```

### Performance Guidelines

#### Query Optimization
```sql
-- Use bind variables instead of literals
SELECT * FROM COMMON.HOTELS WHERE hotel_id = :hotel_id;

-- Use appropriate indexes
-- Create index: CREATE INDEX idx_hotels_city_status ON HOTELS(city, status);
SELECT hotel_id, hotel_name
FROM COMMON.HOTELS
WHERE city = :city
AND status = 'ACTIVE';

-- Use EXISTS instead of IN for subqueries
SELECT h.hotel_id, h.hotel_name
FROM COMMON.HOTELS h
WHERE EXISTS (
    SELECT 1 
    FROM BRIDGE.RESERVATIONS r 
    WHERE r.hotel_id = h.hotel_id
    AND r.status = 'CONFIRMED'
);

-- Limit result sets appropriately
SELECT * FROM (
    SELECT hotel_id, hotel_name, city
    FROM COMMON.HOTELS
    WHERE status = 'ACTIVE'
    ORDER BY hotel_name
) WHERE ROWNUM <= 100;
```

#### Index Design
```sql
-- Single column indexes for primary lookups
CREATE INDEX idx_hotels_status ON COMMON.HOTELS(status);

-- Composite indexes for multi-column searches
CREATE INDEX idx_reservations_guest_status 
ON BRIDGE.RESERVATIONS(guest_email, status, check_in_date);

-- Function-based indexes for computed columns
CREATE INDEX idx_hotels_upper_name 
ON COMMON.HOTELS(UPPER(hotel_name));

-- Partial indexes with WHERE clause (if supported)
CREATE INDEX idx_active_hotels 
ON COMMON.HOTELS(city, state) 
WHERE status = 'ACTIVE';
```

## Testing Procedures

### Unit Testing

#### SQL Unit Tests
```sql
-- Test script template: test_hotel_amenities.sql
SET SERVEROUTPUT ON

DECLARE
    l_hotel_id NUMBER := 1;
    l_amenity_id NUMBER;
    l_count NUMBER;
BEGIN
    -- Test 1: Create amenity
    DBMS_OUTPUT.PUT_LINE('Test 1: Create hotel amenity');
    
    create_hotel_amenity(
        p_hotel_id => l_hotel_id,
        p_amenity_type => 'POOL',
        p_amenity_name => 'Outdoor Swimming Pool',
        p_amenity_id => l_amenity_id
    );
    
    IF l_amenity_id IS NOT NULL THEN
        DBMS_OUTPUT.PUT_LINE('✓ Amenity created successfully: ' || l_amenity_id);
    ELSE
        DBMS_OUTPUT.PUT_LINE('✗ Failed to create amenity');
    END IF;
    
    -- Test 2: Verify amenity exists
    SELECT COUNT(*)
    INTO l_count
    FROM COMMON.HOTEL_AMENITIES
    WHERE amenity_id = l_amenity_id;
    
    IF l_count = 1 THEN
        DBMS_OUTPUT.PUT_LINE('✓ Amenity found in database');
    ELSE
        DBMS_OUTPUT.PUT_LINE('✗ Amenity not found in database');
    END IF;
    
    -- Test 3: Test error handling
    BEGIN
        create_hotel_amenity(
            p_hotel_id => NULL,
            p_amenity_type => 'GYM',
            p_amenity_name => 'Fitness Center',
            p_amenity_id => l_amenity_id
        );
        DBMS_OUTPUT.PUT_LINE('✗ Should have raised error for null hotel_id');
    EXCEPTION
        WHEN OTHERS THEN
            IF SQLCODE = -20001 THEN
                DBMS_OUTPUT.PUT_LINE('✓ Correctly handled null hotel_id');
            ELSE
                DBMS_OUTPUT.PUT_LINE('✗ Unexpected error: ' || SQLERRM);
            END IF;
    END;
    
    -- Cleanup
    DELETE FROM COMMON.HOTEL_AMENITIES WHERE amenity_id = l_amenity_id;
    COMMIT;
    
    DBMS_OUTPUT.PUT_LINE('Test completed');
END;
/
```

```sql
DECLARE

    INTO l_count
```

## Debugging and Troubleshooting

### Common Issues

#### 1. Invalid Objects After Deployment
```sql
-- Check for invalid objects
SELECT object_name, object_type, status
FROM user_objects
WHERE status = 'INVALID'
ORDER BY object_type, object_name;

-- Recompile invalid objects
BEGIN
    FOR rec IN (
        SELECT object_name, object_type
        FROM user_objects
        WHERE status = 'INVALID'
        AND object_type IN ('PROCEDURE', 'FUNCTION', 'PACKAGE', 'PACKAGE BODY')
    ) LOOP
        BEGIN
            EXECUTE IMMEDIATE 'ALTER ' || rec.object_type || ' ' || 
                            rec.object_name || ' COMPILE';
        EXCEPTION
            WHEN OTHERS THEN
                DBMS_OUTPUT.PUT_LINE('Failed to compile ' || rec.object_name || ': ' || SQLERRM);
        END;
    END LOOP;
END;
/
```

#### DBMS_OUTPUT Debugging
```sql
-- Enable output
SET SERVEROUTPUT ON SIZE 1000000

-- Debug procedure
CREATE OR REPLACE PROCEDURE debug_hotel_search(p_city VARCHAR2) IS
    l_count NUMBER;
BEGIN
    DBMS_OUTPUT.PUT_LINE('Starting hotel search for city: ' || p_city);
    
    SELECT COUNT(*)
    INTO l_count
    FROM COMMON.HOTELS
    WHERE city = p_city;
    
    DBMS_OUTPUT.PUT_LINE('Found ' || l_count || ' hotels');
    
    IF l_count = 0 THEN
        DBMS_OUTPUT.PUT_LINE('No hotels found - checking for similar cities');
        
        FOR rec IN (
            SELECT DISTINCT city
            FROM COMMON.HOTELS
            WHERE UPPER(city) LIKE UPPER('%' || p_city || '%')
        ) LOOP
            DBMS_OUTPUT.PUT_LINE('Similar city found: ' || rec.city);
        END LOOP;
    END IF;
END;
/
```

## Best Practices

### Development Best Practices

#### Code Quality
- Write self-documenting code with clear variable names
- Add comments for complex business logic
- Use consistent formatting and indentation
- Follow established naming conventions
- Include error handling in all procedures

#### Version Control
- Create meaningful commit messages
- Keep commits focused on single features
- Use feature branches for all development
- Include rollback scripts with all changes
- Tag releases appropriately

#### Testing
- Test all changes in development environment first
- Create unit tests for stored procedures
- Validate performance impact of changes
- Test rollback procedures before deployment
- Document test cases and results

### Security Best Practices

#### Data Protection
- Never hardcode passwords or sensitive data
- Use bind variables to prevent SQL injection
- Implement proper input validation
- Follow principle of least privilege for grants
- Audit sensitive data access

#### Access Control
- Use role-based security model
- Grant minimum necessary permissions
- Regularly review and audit permissions
- Use service accounts for applications
- Implement proper authentication mechanisms

### Performance Best Practices

#### Query Optimization
- Use appropriate indexes for query patterns
- Avoid SELECT * in production code
- Use bind variables consistently
- Implement proper WHERE clause filtering
- Consider partitioning for large tables

#### Resource Management
- Close cursors properly in PL/SQL
- Use bulk operations for large data sets
- Implement proper connection pooling
- Monitor and tune memory usage
- Plan for data growth and archiving
